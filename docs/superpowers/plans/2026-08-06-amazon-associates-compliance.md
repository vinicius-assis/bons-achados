# Amazon Associates Compliance Fixes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix the two reasons Amazon closed the Associates account (`bonsachados0f-20`) — Amazon links published without the affiliate tag, and public offer cards with no original content — so the site can be reapplied.

**Architecture:** Amazon product entry becomes exclusive to the existing automated hub (`/admin/amazon`, which already builds tag-safe links), removed from the free-text manual form. A new required `note` field is added to the `Highlight` model and threaded through the store, the `/api/admin/highlights` route, all three "Destacar na vitrine" UI flows (Amazon hub, Mercado Livre hub, manual form), and the public vitrine page. Existing Amazon highlights (potentially tag-less) are deleted from the database as a one-off cleanup.

**Tech Stack:** Next.js (App Router) route handlers, Prisma/PostgreSQL (Neon), Vitest, React client components, Tailwind CSS (existing design tokens: `paper`/`ink`/`gold`/`ash`/`alert`).

## Global Constraints

- Spec source: `docs/superpowers/specs/2026-08-06-amazon-associates-compliance-design.md`.
- All `/api/admin/*` routes gate on `isAuthorizedAdminRequest` from `@/lib/adminSession` — unchanged, do not touch that gate.
- `note` minimum length is 15 characters after trimming whitespace, enforced identically on client and server via one shared helper (`src/lib/highlights/note.ts`) — do not duplicate the number `15` anywhere else.
- No test infrastructure exists for React components in this project (`AmazonAdmin.tsx`, `MercadoLivreAdmin.tsx`, `NovoProdutoForm.tsx`, `page.tsx` have no test files today) — tasks touching those files are verified with `npx tsc --noEmit` and `npx eslint`, not new test files.
- Do not modify `/api/admin/postdraft`, `PostDraft`, or the "Selecionar para postar" flows in any admin page — `note` is a `Highlight`-only concept (public vitrine content), out of scope for Telegram/Instagram drafts.
- Do not touch Shopee's link validation (or lack thereof) in `NovoProdutoForm.tsx` — only the Amazon option and the note field are in scope there.

```typescript
import type { Highlight, Marketplace } from "@prisma/client";
import { prisma } from "@/lib/prisma";
import { startOfTodayInBrazil } from "@/lib/date";

export type CreateHighlightInput = {
  marketplace: Marketplace;
  title: string;
  note: string;
  affiliateLink: string;
  image: string;
  price: number;
  oldPrice: number | null;
  discount: number | null;
};

export async function createHighlight(input: CreateHighlightInput): Promise<Highlight> {
  return prisma.highlight.create({ data: input });
}

export async function listTodaysHighlights(): Promise<Highlight[]> {
  return prisma.highlight.findMany({
    where: { createdAt: { gte: startOfTodayInBrazil() } },
    orderBy: { createdAt: "desc" },
    // Explicit select so a future internal-only column added to Highlight
    // isn't silently exposed on the public vitrine page.
    select: {
      id: true,
      marketplace: true,
      title: true,
      note: true,
      affiliateLink: true,
      image: true,
      price: true,
      oldPrice: true,
      discount: true,
      createdAt: true,
    },
  });
}

export async function removeHighlight(id: string): Promise<void> {
  await prisma.highlight.deleteMany({ where: { id } });
}

export async function deleteStaleHighlights(): Promise<number> {
  const result = await prisma.highlight.deleteMany({
    where: { createdAt: { lt: startOfTodayInBrazil() } },
  });
  return result.count;
}
```

```typescript
import { NextRequest, NextResponse } from "next/server";
import type { Marketplace } from "@prisma/client";
import { isAuthorizedAdminRequest } from "@/lib/adminSession";
import { createHighlight, listTodaysHighlights } from "@/lib/highlights/store";
import { isValidNote } from "@/lib/highlights/note";

const VALID_MARKETPLACES: Marketplace[] = ["MERCADO_LIVRE", "AMAZON", "SHOPEE"];

export async function GET(request: NextRequest) {
  if (!isAuthorizedAdminRequest(request)) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const items = await listTodaysHighlights();
  return NextResponse.json({ items });
}

export async function POST(request: NextRequest) {
  if (!isAuthorizedAdminRequest(request)) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const body = await request.json().catch(() => null);
  const marketplace = body?.marketplace;
  const title = typeof body?.title === "string" ? body.title.trim() : "";
  const note = typeof body?.note === "string" ? body.note.trim() : "";
  const affiliateLink = typeof body?.affiliateLink === "string" ? body.affiliateLink.trim() : "";
  const image = typeof body?.image === "string" ? body.image.trim() : "";
  const price = typeof body?.price === "number" ? body.price : NaN;
  const oldPrice = typeof body?.oldPrice === "number" ? body.oldPrice : null;
  const discount = typeof body?.discount === "number" ? body.discount : null;

  if (
    !VALID_MARKETPLACES.includes(marketplace) ||
    !title ||
    !isValidNote(note) ||
    !affiliateLink ||
    !image ||
    Number.isNaN(price) ||
    !Number.isFinite(price) ||
    price <= 0
  ) {
    return NextResponse.json({ error: "invalid_body" }, { status: 400 });
  }

  const created = await createHighlight({
    marketplace,
    title,
    note,
    affiliateLink,
    image,
    price,
    oldPrice,
    discount,
  });

  return NextResponse.json({ id: created.id }, { status: 201 });
}
```

```typescript
    if (!row.title || !row.affiliateLink || !row.image || !row.price.trim() || Number.isNaN(price)) {
      setHighlightStatuses((previous) => ({

    }

      }));
      return;
    }

    setHighlightStatuses((previous) => ({ ...previous, [row.key]: { kind: "saving" } }));
    try {
      const response = await fetch("/api/admin/highlights", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          marketplace: "SHOPEE",

          image: row.image,
          price,
          oldPrice: null,

        }),
      });
      if (!response.ok) {

        }));
        return;
```
